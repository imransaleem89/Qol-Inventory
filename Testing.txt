---
- name: Testing - Open OWA and check inbox
  hosts: Testing
  gather_facts: false

  vars:
    owa_url: "https://10.18.16.53/owa"
    owa_username: "isaleem"
    owa_password: "123123"

  tasks:

    - name: Open Google Chrome with OWA
      ansible.windows.win_shell: |
        $chrome = "${env:ProgramFiles}\Google\Chrome\Application\chrome.exe"

        if (-not (Test-Path $chrome)) {
            $chrome = "${env:ProgramFiles(x86)}\Google\Chrome\Application\chrome.exe"
        }

        if (-not (Test-Path $chrome)) {
            Write-Error "Google Chrome was not found."
            exit 1
        }

        Start-Process $chrome -ArgumentList "{{ owa_url }}"
      args:
        executable: powershell.exe

    - name: Wait for OWA page to load
      ansible.builtin.pause:
        seconds: 5

    - name: Manual CAPTCHA/login required
      ansible.builtin.pause:
        prompt: |
          Chrome has been opened on the target PC.

          1. Enter username: {{ owa_username }}
          2. Enter the password.
          3. Complete the CAPTCHA manually.
          4. Click Login.
          5. When the Inbox is displayed, press ENTER here to continue.

    - name: Wait for inbox to finish loading
      ansible.builtin.pause:
        seconds: 5

    - name: Display next step
      ansible.builtin.debug:
        msg: >
          OWA login completed. The next step is to read the newest
          message subject and received time from the Inbox.
